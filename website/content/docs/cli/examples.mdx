## Browser package

```ts
import { defineConfig } from '@momots/cli';

export default defineConfig({
  build: { target: 'browser' },
});
```

## Bun CLI package

```ts
import { defineConfig } from '@momots/cli';

export default defineConfig({
  build: {
    target: 'bun',
    executables: ['bin.js'],
  },
});
```

## Custom build steps

```ts
import { defineConfig } from '@momots/cli';

export default defineConfig({
  build: {
    scripts: ['lint', 'typecheck', 'types'],
    packages: 'bundle',
  },
});
```

## Bundle specific dependencies

Keep other declared dependencies external and bundle only the listed packages:

```ts
import { defineConfig } from '@momots/cli';

export default defineConfig({
  build: {
    bundle: ['@momots/host'],
  },
});
```

## Test scripts

Run non-browser and browser tests separately from the repository root:

```json
{
  "scripts": {
    "test:unit": "bun test --isolate",
    "test:browser": "playwright test"
  }
}
```

Non-browser tests use `bun:test` and `*.test.ts`; Playwright collects `*.spec.ts`.
Both live beside source files. To keep Bun from collecting Playwright tests, configure `bunfig.toml`:

```toml
[test]
pathIgnorePatterns = ["**/*.spec.ts"]
```

Playwright uses the root `playwright.config.ts`. After installing dependencies, download
the configured Chromium and WebKit browsers from the repository root:

```bash
bunx playwright install chromium webkit
```

Bun builds and serves the pages over HTTP; Playwright Test runs in Node.js.
See the [browser testing tutorial](/docs/cli/tutorial) for file responsibilities and commands.
